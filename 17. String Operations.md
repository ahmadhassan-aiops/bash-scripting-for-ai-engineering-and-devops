# 17. String Operations

```bash
MESSAGE="Learning bash is so much fun"
echo $MESSAGE
# or
echo "${MESSAGE}"
```

## Length

```console
root@ahmadhassan:~# msg="Ahmad"
root@ahmadhassan:~# echo "${msg}"
Ahmad
root@ahmadhassan:~# echo "${#msg}"
5
```

## Concatenation

```console
# rootpath="/etc"
# filepath="/os-release"
# fullpath=$rootpath$filepath
root@ahmadhassan:~# echo $fullpath
/etc/os-release
```

## Uppercase and Lowercase

```console
root@ahmadhassan:~# echo $msg
Ahmad
root@ahmadhassan:~# echo "${msg^^}"   # uppercase
AHMAD
root@ahmadhassan:~# echo "${msg,,}"   # lowercase
ahmad
root@ahmadhassan:~# echo "$msg" | tr [a-z] [A-Z]   # uppercase
AHMAD
root@ahmadhassan:~# echo "$msg" | tr [A-Z] [a-z]   # lowercase
ahmad
```

## Replace

Syntax: `${variable/find/replace}`

```console
root@ahmadhassan:~# echo ${MESSAGE/Learning/Reading}
Reading bash is so much fun
root@ahmadhassan:~# echo ${MESSAGE}
Learning bash is so much fun
```

## Remove a Word from a String

```console
root@ahmadhassan:~# echo ${MESSAGE}
Learning bash is so much fun
root@ahmadhassan:~# echo ${MESSAGE/Learning}
bash is so much fun
```

## Replace a Character

```console
root@ahmadhassan:~# echo ${MESSAGE}
Learning bash is so much fun
root@ahmadhassan:~# echo ${MESSAGE/L/l}
learning bash is so much fun
```

## Substrings

```console
root@ahmadhassan:~# name="Raza"
root@ahmadhassan:~# echo "${name}"
Raza
root@ahmadhassan:~# echo "${name/R/r}"
raza
root@ahmadhassan:~# echo "${name:0:2}"
Ra
root@ahmadhassan:~# echo "${name:0:3}"
Raz
root@ahmadhassan:~# echo "${name:0:4}"
Raza
root@ahmadhassan:~# echo "${name:0:5}"
Raza
root@ahmadhassan:~# echo "${name:1:5}"
aza
root@ahmadhassan:~# echo "${name::-1}"
Raz
root@ahmadhassan:~# echo "${name::-2}"
Ra
root@ahmadhassan:~# echo "${name::-3}"
R
root@ahmadhassan:~# echo "${name::0}"

root@ahmadhassan:~# echo "${name::1}"
```
