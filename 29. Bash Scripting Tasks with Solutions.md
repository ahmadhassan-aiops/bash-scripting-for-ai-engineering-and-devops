# 29. Bash Scripting Tasks with Solutions

## Task 1: Read and Print Input

```bash
#!/bin/bash
echo -n "Enter some text > "
read text
echo "You entered: $text"
```

## Task 2: Read with a Prompt

```bash
#!/bin/bash
read -p "Enter some text > " text
echo "You entered: $text"
```

## Task 3: Read with a Timeout

```bash
#!/bin/bash
echo -n "Hurry up and type something! > "
if read -t 3 response; then
 echo "Great, you made it in time!"
else
 echo "Sorry, you are too slow!"
fi
```

## Task 4: Arithmetic Operations

```bash
#!/bin/bash
first_num=0
second_num=0
read -p "Enter the first number --> " first_num
read -p "Enter the second number -> " second_num
echo "first number + second number = $((first_num + second_num))"
echo "first number - second number = $((first_num - second_num))"
echo "first number * second number = $((first_num * second_num))"
echo "first number / second number = $((first_num / second_num))"
echo "first number % second number = $((first_num % second_num))"
echo "first number raised to the"
echo "power of the second number = $((first_num ** second_num))"
```

## Task 5: Even or Odd

```bash
#!/bin/bash
number=0
read -p "Enter a number > " number
echo "Number is $number"
if [ $((number % 2)) -eq 0 ]; then
 echo "Number is even"
else
 echo "Number is odd"
fi
```

## Task 9: Counter Inside a for Loop

```bash
#!/bin/bash
i=1
for day in Mon Tue Wed Thu Fri
do
 echo "Weekday $((i++)) : $day"
done
```

## Task 10: Classify the Input with case

```bash
#!/usr/bin/bash
read -p "Enter any number: " num
case $num in
 [0-9])
 echo "you entered single number"
 ;;
 [a-z])
 echo "you entered lower case alpha"
 ;;
 [A-Z])
 echo "you entered upper case alpha"
 ;;
 *)
 echo "Unable to identify your input"
 ;;
esac
```
