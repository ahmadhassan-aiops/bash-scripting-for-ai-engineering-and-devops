# 18. Arrays

An **array** is a collection of elements.

```bash
ahmad=(5 2 3 ahmad)
echo "${ahmad[*]}"
ahmad1=(ahmad "Ahmad Hassan")
echo "${ahmad1[*]}"
echo "${ahmad1[@]}"
```

```console
root@ahmadnode2:~# cmd=(2 5 8 52 53 hostname -i)
root@ahmadnode2:~# echo "${cmd[*]}"
2 5 8 52 53 hostname -i
root@ahmadnode2:~# cmd1=(1 2 3 4 5 6 7)
root@ahmadnode2:~# echo "${cmd1[*]}"
1 2 3 4 5 6 7
```

## Indexing

For `cmd1=(1 2 3 4 5 6 7)`:

| Value | 1 | 2 | 3 | 4 | 5 | 6 | 7 |
|---|---|---|---|---|---|---|---|
| Positive index | 0 | 1 | 2 | 3 | 4 | 5 | 6 |
| Negative index | -7 | -6 | -5 | -4 | -3 | -2 | -1 |

## Slicing

```console
root@ahmadnode2:~# echo "${cmd[@]}"
2 5 8 52 53 hostname -i
root@ahmadnode2:~# echo "${cmd[@]:2}"
8 52 53 hostname -i
root@ahmadnode2:~# echo "${cmd[@]:1}"
5 8 52 53 hostname -i
root@ahmadnode2:~# echo "${cmd[@]:1:2}"
5 8
root@ahmadnode2:~# echo "${cmd[@]:1:3}"
5 8 52
root@ahmadnode2:~# echo "${cmd[@]:1:4}"
5 8 52 53
```

## Indexes, Elements and Count

```console
root@ahmadnode2:~# cmd=(2 5 8 52 53 "hostname -i")
root@ahmadnode2:~# echo "${cmd[@]}"
2 5 8 52 53 hostname -i
root@ahmadnode2:~# echo "${!cmd[@]}"    # index values
0 1 2 3 4 5
root@ahmadnode2:~# echo "${cmd[2]}"
8
root@ahmadnode2:~# echo "${#cmd[@]}"    # total number of elements
6
root@ahmadnode2:~# cmd=(2 5 8 52 53 hostname -i)
root@ahmadnode2:~# echo "${#cmd[@]}"
7
root@ahmadnode2:~# echo "${!cmd[@]}"
0 1 2 3 4 5 6
```

## Storing Command Output in an Array

```console
root@ahmadhassan:~# docker -v
Docker version 20.10.14, build a224086

root@ahmadhassan:~# cmd2=$(docker -v)
root@ahmadhassan:~# echo $cmd2
Docker version 20.10.14, build a224086
root@ahmadhassan:~# cmd2=(docker -v)
root@ahmadhassan:~# echo $cmd2
docker
root@ahmadhassan:~# cmd2=( $(docker -v) )
root@ahmadhassan:~# echo $cmd2
Docker
root@ahmadhassan:~# echo "${cmd2[@]}"
Docker version 20.10.14, build a224086
root@ahmadhassan:~# echo "${!cmd2[@]}"
0 1 2 3 4
root@ahmadhassan:~# echo "${#cmd2[@]}"
5
root@ahmadhassan:~# echo "${cmd2[2]}"
20.10.14,
root@ahmadhassan:~# echo "${cmd2[3]}"
build
root@ahmadhassan:~# echo "${cmd2[@]:0:2}"
Docker version
root@ahmadhassan:~# echo "${cmd2[@]:0:3}"
Docker version 20.10.14,
root@ahmadhassan:~# echo "${cmd2[@]::3}"
Docker version 20.10.14,
```

## Reading Input into an Array

Plain `read` stores everything as a single element:

```console
root@ahmadhassan:~# read test
Ahmad Ali Kazim
root@ahmadhassan:~# echo $test
Ahmad Ali Kazim
root@ahmadhassan:~# echo "${test[0]}"
Ahmad Ali Kazim
root@ahmadhassan:~# echo "${test[1]}"

root@ahmadhassan:~# echo "${#test[@]}"
1
root@ahmadhassan:~# echo "${!test[@]}"
0
```

`read -a` stores each word as a separate element:

```console
root@ahmadhassan:~# read -a test1
AHMAD ali KAZIM
root@ahmadhassan:~# echo "${!test1[@]}"
0 1 2
root@ahmadhassan:~# echo "${#test1[@]}"
3
root@ahmadhassan:~# echo "${test1[@]}"
AHMAD ali KAZIM
root@ahmadhassan:~# echo "${test1[2]}"
KAZIM
```

## Custom Indexes

```console
root@ahmadhassan:~# test=([ahmad]=ali [12]=2 [13]=5)
root@ahmadhassan:~# echo "${test[@]}"
ali 2 5
root@ahmadhassan:~# echo "${!test[@]}"
0 12 13
root@ahmadhassan:~# test=([8]=ali [12]=2 [13]=5)
root@ahmadhassan:~# echo "${test[@]}"
ali 2 5
root@ahmadhassan:~# echo "${!test[@]}"
8 12 13
root@ahmadhassan:~# echo "${test[1]}"

root@ahmadhassan:~# echo "${test[12]}"
2
root@ahmadhassan:~# echo "${test[13]}"
5
```

## Associative Arrays (Key/Value)

You need to declare the array first with `declare -A <variable>`.

```console
root@ahmadhassan:~# declare -A mytest
root@ahmadhassan:~# mytest=([fname]="Ahmad" [lname]="Hassan" [15]="AHMAD")
root@ahmadhassan:~# echo "${mytest[@]}"
Ahmad Hassan AHMAD
root@ahmadhassan:~# echo "${#mytest[@]}"
3
root@ahmadhassan:~# echo "${!mytest[@]}"
fname lname 15
root@ahmadhassan:~# echo "${mytest[fname]}"
Ahmad
root@ahmadhassan:~# echo "${mytest[lname]}"
Hassan
```

**Without `declare -A`:**

```console
root@ahmadhassan:~# test1=([one]=1 [two]=2 [three]=3)
root@ahmadhassan:~# echo "${!test1[*]}"
0
root@ahmadhassan:~# echo "${test1[*]}"
3
root@ahmadhassan:~# echo "${test1[one]}"
3
root@ahmadhassan:~# test2=([3]=1 [5]=2 [8]=3)
root@ahmadhassan:~# echo "${test2[@]}"
1 2 3
root@ahmadhassan:~# echo "${!test2[@]}"
3 5 8
```

**With `declare -A`:**

```console
root@ahmadhassan:~# declare -A test3
root@ahmadhassan:~# test3=([one]=1 [two]=2 [three]=3)
root@ahmadhassan:~# echo "${!test3[@]}"
two three one
root@ahmadhassan:~# echo "${#test3[@]}"
3
root@ahmadhassan:~# echo "${test3[@]}"
2 3 1
root@ahmadhassan:~# echo "${test3[one]}"
1
root@ahmadhassan:~# echo "${test3[two]}"
2
root@ahmadhassan:~# echo "${test3[three]}"
3
```
